1"""Core methylation extraction functions for bam2tensor. 

2 

3This module contains the main function for extracting methylation data from 

4BAM files: extract_methylation_data_from_bam(). This function reads aligned 

5bisulfite-sequencing (BS-seq) and enzymatic methylation sequencing (EM-seq) 

6reads and generates a sparse matrix representation of methylation states at 

7each CpG site, with one row per sequenced DNA fragment. 

8 

9The extraction process: 

10 1. Iterates through each chromosome in the genome embedding 

11 2. For each aligned read that passes quality filters: 

12 a. Determines which bisulfite strand the fragment derives from 

13 b. Identifies the CpG sites whose informative base the read covers 

14 c. Extracts the methylation state at each of those CpGs 

15 3. Combines the two mates of a read pair into a single fragment 

16 4. Builds a sparse COO matrix with one row per fragment 

17 

18Methylation State Encoding: 

19 - 1: Methylated (cytosine remained as C after bisulfite treatment) 

20 - 0: Unmethylated (cytosine converted to T by bisulfite treatment) 

21 - -1: No data (indel, SNV, other non-C/T base at the CpG, or the two 

22 mates of a pair disagreeing) 

23 

24Strands and the informative base: 

25 A fragment derives from either the top or the bottom strand of the 

26 original DNA, and both mates of a pair report on that same strand. BAM 

27 stores every read in top-strand orientation, so for a CpG whose C is at 

28 reference position p: 

29 

30 - a top-strand fragment carries the call on the C, at p 

31 (C = methylated, T = unmethylated); 

32 - a bottom-strand fragment carries it on the G, at p + 1 

33 (G = methylated, A = unmethylated). 

34 

35 Each read is therefore decoded at whichever of the two bases is 

36 informative for its fragment's strand, whatever the read's own alignment 

37 orientation. 

38 

39Supported Aligners: 

40 The module supports BAM files from aligners that provide strand information: 

41 - Bismark: Uses the XM call string (Z=methylated CpG, z=unmethylated 

42 CpG) and the XG tag (CT=top strand, GA=bottom strand) 

43 - Biscuit: Uses the YD tag (f=top strand, r=bottom strand) 

44 - bwameth: Uses the YD tag (f=top strand, r=bottom strand), as Biscuit 

45 - gem3/Blueprint: Uses the XB tag (C=top strand, G=bottom strand) 

46 

47Example: 

48 >>> # xdoctest: +SKIP 

49 >>> from bam2tensor.embedding import GenomeMethylationEmbedding 

50 >>> from bam2tensor.functions import extract_methylation_data_from_bam 

51 >>> 

52 >>> # Create genome embedding 

53 >>> embedding = GenomeMethylationEmbedding( 

54 ... genome_name="hg38", 

55 ... expected_chromosomes=["chr1", "chr2"], 

56 ... fasta_source="reference.fa", 

57 ... ) 

58 >>> 

59 >>> # Extract methylation data 

60 >>> result = extract_methylation_data_from_bam( 

61 ... input_bam="sample.bam", 

62 ... genome_methylation_embedding=embedding, 

63 ... quality_limit=20, 

64 ... ) 

65 >>> 

66 >>> print(f"Extracted {result.matrix.shape[0]} reads, {result.matrix.nnz} data points") 

67"""

112def count_non_cpg_retained_xm(xm_tag: str) -> int: 

113 """Count retained non-CpG cytosines in a Bismark XM methylation string. 

114 

115 Bismark's ``XM`` tag encodes per-base methylation context. Uppercase 

116 letters indicate a cytosine that remained as ``C`` in the read 

117 (i.e., was *not* converted by bisulfite/EM-seq treatment). ``H``, 

118 ``X`` and ``U`` correspond to retained cytosines in CHH, CHG and 

119 unknown-context positions respectively. A high count of these on a 

120 single read is a strong signal of incomplete conversion. 

121 

122 Args: 

123 xm_tag: The value of a read's Bismark ``XM`` tag. 

124 

125 Returns: 

126 The count of ``H``, ``X`` and ``U`` characters in ``xm_tag``. 

127 

128 Example: 

129 >>> count_non_cpg_retained_xm("..Z..hhh..HHH..z..") 

130 3 

131 """ 

132 return xm_tag.count("H") + xm_tag.count("X") + xm_tag.count("U")

135def count_non_cpg_retained_reference( 

136 aligned_segment: pysam.AlignedSegment, 

137 is_reverse_parent_strand: bool, 

138 cpg_positions: Container[int] | None = None, 

139) -> int: 

140 """Count retained non-CpG bases validated against the reference. 

141 

142 For a correctly bisulfite- or EM-seq-converted read, every 

143 non-CpG cytosine on the parent strand should have been converted. 

144 On the forward-parent strand that means every non-CpG ``C`` in the 

145 reference should appear as ``T`` in the read; on the reverse-parent 

146 strand every non-CpG ``G`` should appear as ``A``. Positions where 

147 the read still carries the unconverted base *and* the reference 

148 genuinely has a ``C``/``G`` (i.e., the mismatch is not a SNP) count 

149 as retained. 

150 

151 This is a port of the logic in 

152 ``nebiolabs/mark-nonconverted-reads``, re-using the read's existing 

153 ``MD`` tag (via :py:meth:`pysam.AlignedSegment.get_aligned_pairs` 

154 with ``with_seq=True``) instead of requiring a separate reference 

155 FASTA. 

156 

157 Which cytosines are in CpG context — and so exempt — is best taken 

158 from the CpG index via ``cpg_positions``. Without it, the context is 

159 inferred from the neighbouring reference base as reconstructed from 

160 the read, which cannot see a neighbour lying just outside the 

161 alignment: a methylated CpG cut by the end of the read is then 

162 counted as a retained non-CpG cytosine. 

163 

164 Note: 

165 The count is only as good as the ``MD`` tag. bwa-meth writes 

166 ``MD`` relative to its bisulfite-converted reference. For the 

167 directional libraries bwa-meth is designed for, that gives the 

168 exact count, with one blind spot: a read ``C`` over a reference 

169 ``T`` (or ``G`` over ``A``) is a SNP that conversion makes 

170 invisible, so it is counted as retained. For a non-directional 

171 library aligned with bwa-meth the tag does not describe the 

172 read as stored, and retained cytosines are undercounted. 

173 

174 Args: 

175 aligned_segment: A pysam aligned read. Must carry an ``MD`` 

176 tag; BAMs produced by Bismark, Biscuit, bwameth and gem3 

177 all set this tag by default. 

178 is_reverse_parent_strand: ``True`` if the read derives from the 

179 reverse (OB/CTOB) bisulfite parent strand, ``False`` for 

180 the forward (OT/CTOT) strand. 

181 cpg_positions: 0-based reference positions of the ``C`` of 

182 every CpG the read touches, including a CpG whose ``C`` or 

183 ``G`` lies just outside the alignment. If ``None``, CpG 

184 context is inferred from the read's own reconstruction of 

185 the reference. 

186 

187 Returns: 

188 The number of reference-validated retained non-CpG 

189 cytosines (or guanines, for the reverse parent strand). Returns 

190 ``0`` when the read has no sequence or no ``MD`` tag is present. 

191 """ 

192 if aligned_segment.query_sequence is None: 192 ↛ 193line 192 didn't jump to line 193 because the condition on line 192 was never true

193 return 0 

194 

195 try: 

196 pairs = aligned_segment.get_aligned_pairs(matches_only=True, with_seq=True) 

197 except ValueError: 

198 # MD tag missing — cannot validate against reference. 

199 return 0 

200 

201 # Map ref_pos → reference base (uppercase) for CpG-context lookup. 

202 # matches_only=True guarantees query_pos, ref_pos, ref_base are all set. 

203 ref_pos_to_base = {rpos: rb.upper() for _, rpos, rb in pairs} 

204 

205 # On match, pysam returns ref_base uppercase (query matches ref). 

206 # On mismatch (SNP), it returns lowercase. We only care about matches 

207 # where ref is C/G — those are genuine retained, non-converted bases. 

208 target = "G" if is_reverse_parent_strand else "C" 

209 

210 count = 0 

211 for _, rpos, ref_base in pairs: 

212 if ref_base != target: 

213 # Not a match, or ref is not C/G. This rejects SNPs (lowercase) 

214 # and converted positions (read has T/A, match has different base). 

215 continue 

216 # Exclude CpG context. 

217 if cpg_positions is not None: 

218 # From the CpG index: exact, even when the other base of the CpG 

219 # lies outside the read. On the reverse strand the retained base 

220 # is the G, one position right of the CpG's C. 

221 if (rpos - 1 if is_reverse_parent_strand else rpos) in cpg_positions: 

222 continue 

223 elif is_reverse_parent_strand: 

224 # Reverse strand: previous reference base is C. 

225 if ref_pos_to_base.get(rpos - 1) == "C": 

226 continue 

227 else: 

228 # Forward strand: next reference base is G. 

229 if ref_pos_to_base.get(rpos + 1) == "G": 

230 continue 

231 count += 1 

232 

233 return count

236def is_em_overconversion_read( 

237 read_cpg_states: list[int], 

238 min_cpgs: int, 

239) -> bool: 

240 """Identify reads flagged as EM-seq fragment-level over-conversion. 

241 

242 Loyfer et al. (bioRxiv 2026.03.24.713040) report that EM-seq 

243 produces a reproducible ~1–2.5% of multi-CpG fragments that appear 

244 fully unmethylated across every covered CpG, driven by failed TET 

245 protection and subsequent APOBEC hyper-conversion of an entire 

246 molecule. At constitutively methylated loci these reads are purely 

247 technical. Without a per-region methylation prior, the simplest 

248 correction consistent with their observation is: drop reads whose 

249 covered CpGs are all called unmethylated *and* cover at least 

250 ``min_cpgs`` sites (the paper's Fig. 1C regime where the artifact 

251 diverges clearly from WGBS). 

252 

253 This heuristic also drops genuinely fully-unmethylated biological 

254 fragments, so callers should opt in only when the downstream 

255 application can tolerate that trade-off. 

256 

257 Args: 

258 read_cpg_states: Per-CpG methylation state values for a single 

259 read, in column-order, using the bam2tensor encoding 

260 (``1``=methylated, ``0``=unmethylated, ``-1``=no data). 

261 min_cpgs: Minimum number of covered CpGs required to apply the 

262 filter. Reads with fewer covered CpGs are never flagged. 

263 

264 Returns: 

265 ``True`` when the read has at least ``min_cpgs`` covered CpGs 

266 and every covered CpG is called unmethylated (value ``0``). 

267 ``-1`` (no-data) values do not count as unmethylated. 

268 

269 Example: 

270 >>> is_em_overconversion_read([0, 0, 0], min_cpgs=3) 

271 True 

272 >>> is_em_overconversion_read([0, 0, 1], min_cpgs=3) 

273 False 

274 >>> is_em_overconversion_read([0, 0], min_cpgs=3) 

275 False 

276 """ 

277 if len(read_cpg_states) < min_cpgs: 

278 return False 

279 return all(state == 0 for state in read_cpg_states)

282def detect_aligner(input_bam: str, sample_size: int = 1000) -> str: 

283 """Detect the aligner used to produce a BAM file by checking read tags. 

284 

285 Samples up to ``sample_size`` reads from the BAM file and checks for 

286 aligner-specific tags: ``XM`` (Bismark), ``YD`` (Biscuit/bwameth), 

287 or ``XB`` (gem3/Blueprint). 

288 

289 Args: 

290 input_bam: Path to an indexed BAM file. 

291 sample_size: Maximum number of reads to examine. Only primary, 

292 non-duplicate, non-QC-failed reads are considered. 

293 

294 Returns: 

295 A human-readable string identifying the detected aligner, such as 

296 ``"Bismark (XM tag)"`` or ``"Unknown"``. 

297 """ 

298 try: 

299 bam = pysam.AlignmentFile(input_bam, "rb", require_index=True) # type: ignore 

300 except (FileNotFoundError, ValueError): 

301 return "Unknown (could not open BAM)" 

302 

303 checked = 0 

304 for read in bam.fetch(): 

305 if read.flag & _SKIP_FLAGS: 

306 continue 

307 if read.has_tag("XM"): 

308 bam.close() 

309 return "Bismark (XM tag)" 

310 if read.has_tag("YD"): 

311 bam.close() 

312 return "Biscuit/bwameth (YD tag)" 

313 if read.has_tag("XB"): 

314 bam.close() 

315 return "gem3/Blueprint (XB tag)" 

316 checked += 1 

317 if checked >= sample_size: 

318 break 

319 

320 bam.close() 

321 return "Unknown (no XM/YD/XB tags found)"

324def check_chromosome_overlap( 

325 bam_references: tuple[str, ...] | list[str], 

326 embedding_chromosomes: list[str], 

327) -> None: 

328 """Check that BAM and embedding chromosome names overlap. 

329 

330 Detects the common mistake where a BAM file uses one chromosome naming 

331 convention (e.g., ``1``, ``2``, ``3`` from Ensembl) while the reference 

332 embedding uses another (e.g., ``chr1``, ``chr2``, ``chr3`` from UCSC). 

333 If no overlap is found but adding or stripping a ``chr`` prefix would 

334 create overlap, raises a clear error with guidance. 

335 

336 Args: 

337 bam_references: Tuple or list of chromosome/contig names from the 

338 BAM file header (e.g., from ``pysam.AlignmentFile.references``). 

339 embedding_chromosomes: List of chromosome names from the genome 

340 embedding (i.e., the expected chromosomes). 

341 

342 Raises: 

343 ValueError: If there is no overlap between BAM and embedding 

344 chromosome names, with a diagnostic message indicating 

345 whether a ``chr`` prefix mismatch is the likely cause. 

346 """ 

347 bam_set = set(bam_references) 

348 emb_set = set(embedding_chromosomes) 

349 

350 overlap = bam_set & emb_set 

351 if overlap: 

352 return 

353 

354 # Try adding "chr" prefix to BAM references 

355 bam_with_chr = {f"chr{name}" for name in bam_set} 

356 if bam_with_chr & emb_set: 

357 bam_example = next(iter(bam_set - emb_set)) 

358 emb_example = f"chr{bam_example}" 

359 raise ValueError( 

360 f"Chromosome name mismatch: BAM file uses '{bam_example}' " 

361 f"but the reference embedding expects '{emb_example}'. " 

362 f"Your BAM file appears to use Ensembl-style names (no 'chr' prefix) " 

363 f"while the embedding uses UCSC-style names ('chr' prefix). " 

364 f"Use --expected-chromosomes with matching names " 

365 f"(e.g., --expected-chromosomes '1,2,3,...')." 

366 ) 

367 

368 # Try stripping "chr" prefix from BAM references 

369 bam_stripped = {name[3:] for name in bam_set if name.startswith("chr")} 

370 if bam_stripped & emb_set: 

371 bam_example = next( 

372 name for name in bam_set if name.startswith("chr") and name[3:] in emb_set 

373 ) 

374 emb_example = bam_example[3:] 

375 raise ValueError( 

376 f"Chromosome name mismatch: BAM file uses '{bam_example}' " 

377 f"but the reference embedding expects '{emb_example}'. " 

378 f"Your BAM file uses UCSC-style names ('chr' prefix) " 

379 f"while the embedding uses Ensembl-style names (no 'chr' prefix). " 

380 f"Use --expected-chromosomes with matching names " 

381 f"(e.g., --expected-chromosomes 'chr1,chr2,chr3,...')." 

382 ) 

383 

384 # No overlap even with prefix changes -- completely different names 

385 bam_sample = sorted(bam_set)[:3] 

386 emb_sample = sorted(emb_set)[:3] 

387 raise ValueError( 

388 f"No overlapping chromosome names between BAM file and reference " 

389 f"embedding. BAM contains: {bam_sample}... " 

390 f"Embedding expects: {emb_sample}... " 

391 f"Ensure the reference FASTA matches the genome used for alignment " 

392 f"and that --expected-chromosomes is set correctly." 

393 )

398 

399 Observations are held in compact typed arrays rather than Python lists

566def _bismark_is_bottom_strand( 

567 aligned_segment: pysam.AlignedSegment, xm_tag: str 

568) -> bool: 

569 """Tell whether a Bismark alignment derives from the bottom strand. 

570 

571 Bismark records the strand of the genome a read (pair) aligned to in the 

572 ``XG`` tag: ``CT`` for the original top strand, ``GA`` for the original 

573 bottom strand. Both mates of a pair carry the same value. Bismark writes 

574 each methylation call on the base that carries it, so calls for a 

575 top-strand alignment sit on the reference ``C`` of a CpG and calls for a 

576 bottom-strand alignment sit on the ``G``. 

577 

578 If ``XG`` is absent (a Bismark-style BAM from another tool), the strand 

579 is inferred from the read bases the calls sit on: ``C``/``T`` for the 

580 top strand, ``G``/``A`` for the bottom strand.

634def _merge_mate_calls( 

635 cols: list[int], 

636 data: list[int], 

637 mate_cols: list[int], 

638 mate_data: list[int], 

639) -> tuple[list[int], list[int]]: 

640 """Combine the CpG calls of a fragment's two mates. 

641 

642 Mates that overlap report some CpGs twice. Those are the same bases of 

643 the same molecule read twice, so they are recorded once: 

644 

645 - if both mates make the same call, that call is kept; 

646 - if only one mate has a call (the other has ``-1``), that call is kept; 

647 - if the mates disagree (one methylated, one unmethylated), at least one 

648 is a sequencing error and there is no telling which, so the CpG is 

649 recorded as ``-1`` (no data). 

650 

651 Args: 

652 cols: Matrix columns called by the first mate, ascending. 

653 data: Methylation state for each entry of ``cols``. 

654 mate_cols: Matrix columns called by the second mate, ascending. 

655 mate_data: Methylation state for each entry of ``mate_cols``. 

656 

657 Returns: 

658 The merged columns, ascending, and the state for each. 

659 

660 Example: 

661 >>> _merge_mate_calls([3, 4], [1, 0], [7, 8], [1, 1]) 

662 ([3, 4, 7, 8], [1, 0, 1, 1]) 

663 >>> _merge_mate_calls([3, 4, 5], [1, -1, 0], [4, 5, 6], [1, 1, 0]) 

664 ([3, 4, 5, 6], [1, 1, -1, 0]) 

665 """ 

666 # Common case: the mates do not overlap at any CpG. 

667 if cols[-1] < mate_cols[0]: 

668 return cols + mate_cols, data + mate_data 

669 if mate_cols[-1] < cols[0]: 

670 return mate_cols + cols, mate_data + data 

671 

672 merged = dict(zip(cols, data)) 

673 for col, state in zip(mate_cols, mate_data): 

674 seen = merged.get(col) 

675 if seen is None or seen == -1: 

676 merged[col] = state 

677 elif state != -1 and state != seen: 

678 merged[col] = -1 

679 merged_cols = sorted(merged) 

680 return merged_cols, [merged[col] for col in merged_cols]

1035) -> ExtractionResult: 

1036 """Extract per-fragment CpG methylation data from a BAM file. 

1037 

1038 Parses a bisulfite-sequencing or EM-seq BAM file and extracts methylation 

1039 states at each CpG site covered by aligned reads. Returns a sparse matrix 

1040 where each row represents one sequenced DNA fragment and each column 

1041 represents a CpG site. 

1042 

1043 **Rows are fragments.** The two mates of a read pair come from the same 

1044 molecule, so their calls are combined into a single row; a single-end 

1045 read, or a read whose mate is unusable, is a row on its own. Where the 

1046 mates overlap, a CpG they both cover is recorded once: the shared call 

1047 if they agree, the available call if only one mate has one, and ``-1`` 

1048 if they disagree. Mates are combined when both pass the read filters 

1049 and map to the same chromosome within 100 kb of each other; otherwise 

1050 each is reported as its own fragment. 

1051 

1052 The function applies several filters to reads: 

1053 - Mapping quality must be >= quality_limit (default 20) 

1054 - Duplicate reads are excluded 

1055 - QC-failed reads are excluded 

1056 - Secondary and supplementary alignments are excluded 

1057 - Unmapped reads are excluded (including unmapped mates placed at 

1058 their mapped mate's position) 

1059 - Reads without a recognised strand tag are excluded 

1060 

1061 Two additional, opt-in filters are available: 

1062 - Non-converted reads (``filter_non_converted``): drops fragments 

1063 in which a read has too many retained non-CpG cytosines, the 

1064 hallmark of incomplete bisulfite/EM-seq conversion. Ports the 

1065 per-read logic of ``nebiolabs/mark-nonconverted-reads``; because 

1066 incomplete conversion affects the whole molecule, a read that 

1067 fails takes its mate with it. For Biscuit/bwameth/gem3 reads 

1068 this needs the ``MD`` tag; reads without one cannot be 

1069 evaluated, are kept, and trigger a ``RuntimeWarning`` reporting 

1070 how many were affected. 

1071 - EM-seq fragment-level over-conversion 

1072 (``filter_em_overconversion``): drops fragments whose covered 

1073 CpGs (across both mates) are all called unmethylated, a 

1074 heuristic for the EM-seq artifact described by Loyfer et al. 

1075 (bioRxiv 2026.03.24.713040). 

1076 

1077 Every read is decoded at the base that is informative for the strand 

1078 its fragment derives from: the ``C`` of a CpG for a top-strand 

1079 fragment, the ``G`` for a bottom-strand fragment (BAM stores reads in 

1080 top-strand orientation). This is independent of the read's own 

1081 alignment orientation, so both mates of a pair are used. Two tag 

1082 dialects are supported, detected automatically per read: 

1083 

1084 **Bismark** (XM tag present): 

1085 1. Takes the fragment's strand from the XG tag (CT = top, GA = 

1086 bottom) 

1087 2. Uses binary search to find the CpGs whose informative base the 

1088 read covers 

1089 3. Reads the XM call string there: Z=methylated(1), 

1090 z=unmethylated(0), anything else=-1 

1091

1093 1. Takes the fragment's strand from the tag (YD f / XB C = top, 

1094 YD r / XB G = bottom) 

1095 2. Uses binary search to find the CpGs whose informative base the 

1096 read covers

1100 

1101 Args: 

1102 input_bam: Path to the input BAM file. The file must be indexed 

1103 (have a corresponding .bam.bai file).

1105 that defines the CpG site positions and provides coordinate 

1106 conversion methods. 

1107 quality_limit: Minimum mapping quality (MAPQ) threshold for reads. 

1108 Reads with MAPQ below this value are skipped. Default is 20, 

1109 which excludes reads mapping to multiple locations equally well. 

1110 filter_non_converted: If True, drop fragments in which a read 

1111 carries at least ``non_converted_threshold`` retained non-CpG 

1112 cytosines, a signature of incomplete bisulfite/EM-seq 

1113 conversion. Default False.

1115 cytosines required for the non-converted filter to drop a 

1116 read. Matches the NEB ``mark-nonconverted-reads`` default of 

1117 3. 

1118 filter_em_overconversion: If True, drop fragments whose covered 

1119 CpGs are all called unmethylated and cover at least 

1120 ``em_overconversion_min_cpgs`` sites — the Loyfer et al. 

1121 EM-seq fragment-level over-conversion heuristic. Default 

1122 False. 

1123 em_overconversion_min_cpgs: Minimum covered CpG count required 

1124 before the over-conversion filter will drop a fragment. Matches 

1125 the regime in Loyfer et al. Fig. 1C where the EM-seq 

1126 artifact is clearly separable from WGBS.

1128 worker. Default 1 (in-process, no IPC overhead). Values 

1129 above 1 fan out per-chromosome extraction across worker

1133 in. Each worker is sent only the CpG positions of the 

1134 chromosome it is processing, not the whole embedding. 

1135 Recommended: leave at 1 for small BAMs (worker 

1136 startup dominates); set to the number of physical cores 

1137 for whole-genome WGBS BAMs. Workers are started with the 

1138 ``spawn`` method, so a script that passes ``threads > 1`` 

1139 must do so from under ``if __name__ == "__main__":`` 

1140 (standard :mod:`multiprocessing` practice). 

1141 verbose: If True, display a progress bar and print the total read 

1142 count. Useful for monitoring progress on large files. 

1143 debug: If True, print the calls decoded from every read and 

1144 each filter decision. Significantly slower. 

1145 

1146 Returns: 

1147 An ExtractionResult named tuple with two fields: 

1148 

1149 - **matrix**: A scipy.sparse.coo_matrix with shape 

1150 (n_fragments, n_cpg_sites) where n_fragments is the number of 

1151 fragments that passed filters and covered at least one CpG 

1152 site, n_cpg_sites is 

1153 genome_methylation_embedding.total_cpg_sites, and values are: 

1154 1 (methylated), 0 (unmethylated), -1 (no data). Within a row, 

1155 CpGs are in ascending column order. Values are stored as int64. 

1156 The matrix uses COO format for efficient construction; convert 

1157 to CSR (tocsr()) for row slicing or CSC (tocsc()) for column 

1158 slicing. 

1159 - **tlen**: A 1-D numpy int32 array of shape (n_fragments,) 

1160 containing the length of each fragment (the absolute value of 

1161 the BAM TLEN field). Values are 0 for single-end reads or reads 

1162 with unmapped mates. 

1163 

1164 Raises: 

1165 FileNotFoundError: If the BAM file index (.bam.bai) is missing. 

1166 The BAM file must be indexed with `samtools index`. 

1167 ValueError: If ``threads`` is less than 1. 

1168 

1169 Example: 

1170 >>> # xdoctest: +SKIP 

1171 >>> from bam2tensor.embedding import GenomeMethylationEmbedding 

1172 >>> from bam2tensor.functions import extract_methylation_data_from_bam 

1173 >>> import scipy.sparse 

1174 >>> 

1175 >>> # Create embedding (or load from cache) 

1176 >>> embedding = GenomeMethylationEmbedding( 

1177 ... genome_name="hg38", 

1178 ... expected_chromosomes=["chr1", "chr22"], 

1179 ... fasta_source="GRCh38.fa", 

1180 ... ) 

1181 >>> 

1182 >>> # Extract methylation data 

1183 >>> result = extract_methylation_data_from_bam( 

1184 ... input_bam="sample.bam", 

1185 ... genome_methylation_embedding=embedding, 

1186 ... quality_limit=30, # Stricter quality filter 

1187 ... verbose=True, 

1188 ... ) 

1189 >>> 

1190 >>> # Analyze results 

1191 >>> print(f"Fragments with CpG data: {result.matrix.shape[0]:,}") 

1192 >>> print(f"Total CpG sites: {result.matrix.shape[1]:,}") 

1193 >>> print(f"Data points: {result.matrix.nnz:,}") 

1194 >>> 

1195 >>> # Save to file 

1196 >>> scipy.sparse.save_npz("sample.methylation.npz", result.matrix) 

1197 

1198 Note: 

1199 Rows are ordered by chromosome, in the order chromosomes appear in 

1200 genome_methylation_embedding.cpg_sites_dict, and within a 

1201 chromosome by where each fragment's first CpG-covering read 

1202 appears in the (coordinate-sorted) BAM. 

1203 

1204 See Also: 

1205 GenomeMethylationEmbedding: For creating the genome embedding. 

1206 scipy.sparse.coo_matrix: Documentation on the output format. 

1207 """ 

1208 if threads < 1: 

1209 raise ValueError(f"threads must be >= 1, got {threads}") 

1210 

1211 # Open once on the parent side to surface a clean error if the index 

1212 # is missing, to validate chromosome-name overlap, and to read the 

1213 # mapped-read count for verbose output. Workers open their own 

1214 # handles. Sequential mode passes pysam_threads=4 (the historical 

1215 # setting) to its handle; parallel mode uses 1 per worker since each 

1216 # worker is its own process. The parent handle only touches metadata 

1217 # so threads=1 here is sufficient. 

1218 try: 

1219 input_bam_object = pysam.AlignmentFile( # type: ignore # pylint: disable=no-member 

1220 input_bam, "rb", require_index=True, threads=1 

1221 ) 

1222 except FileNotFoundError as exc: 

1223 raise FileNotFoundError(f"Index missing for bam file?: {input_bam}") from exc 

1224 

1225 check_chromosome_overlap( 

1226 bam_references=input_bam_object.references, 

1227 embedding_chromosomes=list(genome_methylation_embedding.cpg_sites_dict.keys()), 

1228 ) 

1229 if verbose: 

1230 print(f"\tTotal reads: {input_bam_object.mapped:,}\n") 

1231 input_bam_object.close() 

1232 

1233 n_cols = genome_methylation_embedding.total_cpg_sites 

1234 

1235 config = _ExtractionConfig( 

1236 pysam_threads=4 if threads == 1 else 1, 

1237 quality_limit=quality_limit, 

1238 filter_non_converted=filter_non_converted, 

1239 non_converted_threshold=non_converted_threshold, 

1240 filter_em_overconversion=filter_em_overconversion, 

1241 em_overconversion_min_cpgs=em_overconversion_min_cpgs, 

1242 debug=debug, 

1243 col_typecode="i" if n_cols <= np.iinfo(np.int32).max else "q", 

1244 ) 

1245 

1246 # One task per chromosome, in cpg_sites_dict order: the row order of the 

1247 # output follows this order in both the sequential and parallel paths. 

1248 chrom_tasks: list[_ChromTask] = [ 

1249 (chrom, cpg_sites, genome_methylation_embedding.chromosome_offsets[chrom]) 

1250 for chrom, cpg_sites in genome_methylation_embedding.cpg_sites_dict.items() 

1251 ] 

1252 

1253 def _iter_chrom_results() -> Iterator[tuple[str, _ChromExtractionResult]]: 

1254 if threads == 1: 

1255 for chrom, cpg_sites, col_offset in chrom_tasks: 

1256 yield chrom, _extract_chromosome( 

1257 input_bam=input_bam, 

1258 chrom=chrom, 

1259 cpg_sites=cpg_sites, 

1260 col_offset=col_offset, 

1261 config=config, 

1262 ) 

1263 return 

1264 # Spawn is the safe start method on macOS (fork + pysam can hang 

1265 # because pysam holds C-level locks across the fork). Spawn is 

1266 # also the default on macOS for Python 3.8+, but be explicit. 

1267 ctx = mp.get_context("spawn") 

1268 with ProcessPoolExecutor( 

1269 max_workers=threads, 

1270 mp_context=ctx, 

1271 initializer=_worker_init, 

1272 initargs=(input_bam, config), 

1273 ) as executor: 

1274 # map() preserves submission order — required so chromosomes are 

1275 # concatenated in the same order as the sequential path. The 

1276 # executor pickles tasks a few at a time as workers free up, so 

1277 # only the chromosomes in flight are ever serialised at once. 

1278 yield from zip( 

1279 (task[0] for task in chrom_tasks), 

1280 executor.map(_worker_extract_chromosome, chrom_tasks), 

1281 ) 

1282 

1283 # Per-chromosome chunks, concatenated once at the end. 

1284 col_chunks: list[np.ndarray] = [] 

1285 data_chunks: list[np.ndarray] = [] 

1286 cpgs_per_fragment_chunks: list[np.ndarray] = [] 

1287 tlen_chunks: list[np.ndarray] = [] 

1288 reads_without_md = 0 

1289 

1290 for chrom, chrom_result in tqdm( 

1291 _iter_chrom_results(), total=len(chrom_tasks), disable=not verbose 

1292 ): 

1293 if chrom_result.missing: 

1294 if verbose: 

1295 tqdm.write(f"Chromosome {chrom} not found in BAM file, skipping.") 

1296 continue 

1297 reads_without_md += chrom_result.reads_without_md 

1298 if len(chrom_result.cpgs_per_fragment) == 0: 

1299 continue 

1300 col_chunks.append(chrom_result.cols) 

1301 data_chunks.append(chrom_result.data) 

1302 cpgs_per_fragment_chunks.append(chrom_result.cpgs_per_fragment) 

1303 tlen_chunks.append(chrom_result.tlens) 

1304 

1305 if reads_without_md: 

1306 warnings.warn( 

1307 f"{reads_without_md:,} read(s) in {input_bam} have no MD tag, so the " 

1308 "non-converted filter could not evaluate them and they were kept. " 

1309 "Add MD tags (e.g. with `samtools calmd`) to apply the filter to " 

1310 "these reads.", 

1311 RuntimeWarning, 

1312 stacklevel=2, 

1313 ) 

1314 

1315 def _concatenate(chunks: list[np.ndarray], dtype: type) -> np.ndarray: 

1316 # Concatenate straight into the final dtype, then drop the chunks so 

1317 # their memory is released before the next array is built. 

1318 if not chunks: 

1319 return np.empty(0, dtype=dtype) 

1320 combined: np.ndarray = np.concatenate(chunks, dtype=dtype) 

1321 chunks.clear() 

1322 return combined 

1323 

1324 cpgs_per_fragment = _concatenate(cpgs_per_fragment_chunks, np.int32) 

1325 n_fragments = len(cpgs_per_fragment) 

1326 

1327 # Build the index arrays directly in the dtype scipy will choose for a 

1328 # matrix of this shape, so it adopts them instead of converting copies. 

1329 index_dtype = scipy.sparse.get_index_dtype(maxval=max(n_fragments, n_cols)) 

1330 # Fragments were emitted in order, so each fragment's row index simply 

1331 # repeats once per CpG it covers. 

1332 coo_row = np.repeat(np.arange(n_fragments, dtype=index_dtype), cpgs_per_fragment) 

1333 coo_col = _concatenate(col_chunks, index_dtype) 

1334 # int64 values keep the saved .npz identical to earlier releases, which 

1335 # built the matrix from Python ints. 

1336 coo_data = _concatenate(data_chunks, np.int64) 

1337 tlen_array = _concatenate(tlen_chunks, np.int32) 

1338 

1339 if debug: 

1340 print("Debug info for coo_matrix dimensions:") 

1341 print(f"\tcoo_row: {len(coo_row):,}") 

1342 if len(coo_row): 1342 ↛ 1344line 1342 didn't jump to line 1344 because the condition on line 1342 was always true

1343 print(f"\tcoo row max: {int(coo_row.max()):,}") 

1344 print(f"\tcoo_col: {len(coo_col):,}") 

1345 if len(coo_col): 1345 ↛ 1347line 1345 didn't jump to line 1347 because the condition on line 1345 was always true

1346 print(f"\tcoo col max: {int(coo_col.max()):,}") 

1347 print(f"\tcoo_data: {len(coo_data):,}") 

1348 print(f"\ttotal_cpg_sites: {n_cols:,}") 

1349 

1350 sparse_matrix = scipy.sparse.coo_matrix( 

1351 (coo_data, (coo_row, coo_col)), 

1352 shape=(n_fragments, n_cols), 

1353 ) 

1354 

1355 assert sparse_matrix.shape[1] == n_cols 

1356 

1357 return ExtractionResult(matrix=sparse_matrix, tlen=tlen_array)
